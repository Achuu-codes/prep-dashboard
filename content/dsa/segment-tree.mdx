### When to reach for this

- **"Point update, prefix or range sum"** → Fenwick. Shortest code, smallest constant, and it is all you need for sums and XORs.
- **"Range min / max / gcd with updates"** → segment tree. Fenwick cannot do these: it relies on the operation being **invertible**.
- **"Range update *and* range query"** (add v to `[l, r]`, then ask for a sum) → segment tree with lazy propagation.
- **"Count inversions / how many smaller to the right / order statistics"** → Fenwick over **compressed values**, swept in one direction.
- **"Values up to `10^9` but only `10^5` of them"** → coordinate compression first: `sort`, `unique`, `lower_bound`.
- If there are no updates, you do not need either structure — prefix sums or a sparse table are strictly better.

### Templates

```cpp
// Fenwick / BIT. 1-INDEXED: index 0 makes the loop spin forever (0 & -0 == 0).
struct Fenwick {
    int n;
    vector<long long> t;
    explicit Fenwick(int n) : n(n), t(n + 1, 0) {}
    void add(int i, long long v) { for (; i <= n; i += i & -i) t[i] += v; }
    long long sum(int i) {                      // prefix [1..i]
        long long s = 0;
        for (; i > 0; i -= i & -i) s += t[i];
        return s;
    }
    long long range(int l, int r) { return sum(r) - sum(l - 1); }
};
```

```cpp
// Inversions: compress, then sweep right-to-left counting smaller-seen-later
long long inversions(vector<int> a) {
    vector<int> s = a;
    sort(s.begin(), s.end());
    s.erase(unique(s.begin(), s.end()), s.end());        // compression
    Fenwick f(s.size());
    long long inv = 0;
    for (int i = (int)a.size() - 1; i >= 0; --i) {
        int r = lower_bound(s.begin(), s.end(), a[i]) - s.begin() + 1;  // 1-based
        inv += f.sum(r - 1);                             // strictly smaller
        f.add(r, 1);
    }
    return inv;
}
```

```cpp
// Iterative segment tree, 2n memory. Any associative op; here max.
// query is HALF-OPEN: [l, r).
struct SegTree {
    int n;
    vector<int> t;
    explicit SegTree(int n) : n(n), t(2 * n, INT_MIN) {}
    void set(int i, int v) {
        for (t[i += n] = v; i > 1; i >>= 1) t[i >> 1] = max(t[i], t[i ^ 1]);
    }
    int query(int l, int r) {
        int res = INT_MIN;
        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) res = max(res, t[l++]);
            if (r & 1) res = max(res, t[--r]);
        }
        return res;
    }
};
```

```cpp
// Lazy propagation: range add, range sum. 4n memory, recursive, inclusive [l,r].
struct LazySeg {
    int n;
    vector<long long> t, lz;
    explicit LazySeg(int n) : n(n), t(4 * n, 0), lz(4 * n, 0) {}

    void apply(int node, int l, int r, long long v) {
        t[node] += v * (r - l + 1);            // SUM scales by length
        lz[node] += v;                         // (for max it would not)
    }
    void push(int node, int l, int r) {
        if (!lz[node]) return;
        int m = (l + r) / 2;
        apply(node << 1,     l,     m, lz[node]);
        apply(node << 1 | 1, m + 1, r, lz[node]);
        lz[node] = 0;
    }
    void update(int node, int l, int r, int ql, int qr, long long v) {
        if (qr < l || r < ql) return;                        // disjoint
        if (ql <= l && r <= qr) { apply(node, l, r, v); return; }   // covered
        push(node, l, r);                                    // push BEFORE descending
        int m = (l + r) / 2;
        update(node << 1, l, m, ql, qr, v);
        update(node << 1 | 1, m + 1, r, ql, qr, v);
        t[node] = t[node << 1] + t[node << 1 | 1];           // recombine AFTER
    }
    long long query(int node, int l, int r, int ql, int qr) {
        if (qr < l || r < ql) return 0;                      // identity element
        if (ql <= l && r <= qr) return t[node];
        push(node, l, r);
        int m = (l + r) / 2;
        return query(node << 1, l, m, ql, qr)
             + query(node << 1 | 1, m + 1, r, ql, qr);
    }
};
```

```cpp
// k-th smallest from a Fenwick of counts, by binary lifting: O(log n)
int kth(Fenwick& f, long long k) {
    int pos = 0;
    for (int pw = 1 << (31 - __builtin_clz(f.n)); pw; pw >>= 1)
        if (pos + pw <= f.n && f.t[pos + pw] < k) { pos += pw; k -= f.t[pos]; }
    return pos + 1;
}
```

### Complexity

| Structure | Build | Update | Query | Space |
| --- | --- | --- | --- | --- |
| Fenwick | `O(n)` | `O(log n)` | `O(log n)` | `O(n)` |
| Segment tree (iterative) | `O(n)` | `O(log n)` | `O(log n)` | `2n` |
| Lazy segment tree | `O(n)` | `O(log n)` range | `O(log n)` range | `4n` |
| Fenwick `kth` by lifting | — | — | `O(log n)` | `O(n)` |
| Prefix sums (no updates) | `O(n)` | not supported | `O(1)` | `O(n)` |

### Pitfalls

- **Fenwick is 1-indexed.** Calling `add(0, v)` makes `i += i & -i` loop forever, because `0 & -0` is 0.
- Fenwick needs an **invertible** operation. `range(l, r) = sum(r) - sum(l-1)` is meaningless for max — that needs a segment tree.
- Size the recursive tree `4n` and the iterative one `2n`. `2n` for the recursive version overflows on non-power-of-two `n`.
- **The iterative query is half-open `[l, r)`; the lazy one above is inclusive `[l, r]`.** Mixing conventions between the two is the most common bug in this topic. Pick one per solution and write it in a comment.
- Lazy: `push` before descending, recombine **after** the recursive calls. Reversing those two lines gives stale parents.
- `apply` scales by segment length for **sum** but not for min/max. Copying a sum template into a max problem silently multiplies values.
- The disjoint-case return value must be the operation's **identity**: 0 for sum, `INT_MIN` for max, `INT_MAX` for min — never 0 for all three.
- Use `long long` for sums: `10^5` values of `10^9` overflows `int` immediately.
- Coordinate compression: `sort` then `unique` then `erase`, and remember the `+1` when feeding a 1-indexed Fenwick.
- Inversions can be counted sweeping either direction, but the query changes (smaller-to-the-right versus greater-already-inserted). Choose one and be consistent.
- If the problem has no updates, a segment tree is an expensive way to write a prefix-sum array.
